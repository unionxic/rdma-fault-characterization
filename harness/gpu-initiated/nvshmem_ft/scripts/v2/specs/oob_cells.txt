# B: F2a variants with the device bounds check (BOUNDS=1); PE0 puts one 256 KiB window to the wrong
# place at iteration 4. exact = the next object (sbuf); gap = the freed hole after dbuf; straddle =
# crosses dbuf's end into sbuf; tail = past the last object.

F2a timeout BOUNDS=1 OOB=tail ITERS=10 CORRUPT_AT=4 TAG=b_tail
